cmake_minimum_required(VERSION 3.20)
project(BarrierTrainer LANGUAGES CXX)
find_package(Torch REQUIRED)
add_executable(barrier_train main.cpp)
target_compile_features(barrier_train PRIVATE cxx_std_20)
target_link_libraries(barrier_train PRIVATE "${TORCH_LIBRARIES}")
if(MSVC)
 target_compile_options(barrier_train PRIVATE /EHsc /bigobj /utf-8)
 file(GLOB TORCH_DLLS "${TORCH_INSTALL_PREFIX}/lib/*.dll")
 add_custom_command(TARGET barrier_train POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${TORCH_DLLS} $<TARGET_FILE_DIR:barrier_train> COMMAND_EXPAND_LISTS)
endif()
